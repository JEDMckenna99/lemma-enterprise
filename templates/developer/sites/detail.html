Quick Start

Add authentication to your site:

JavaScript 
const wallet = new LemmaWallet();

// Check if user is authenticated
const result = await wallet.checkAuth();

if (result.authenticated) {
    // User is signed in
    const ppid = result.ppid;
    
    // Send PPID to your backend to identify user
    await fetch('/api/auth', {
        method: 'POST',
        body: JSON.stringify({ ppid })
    });
} else {
    // Show sign-in button
    wallet.startRedirectFlow({
        returnUrl: window.location.href
    });
}
