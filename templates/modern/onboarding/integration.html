⚡ Quick Start (2 minutes)

The fastest way to add human verification to your site:

<script src="https://cdn.jsdelivr.net/npm/@lemma-network/verifier@latest"></script> <script> // Initialize Lemma const lemma = new LemmaVerifier({ apiKey: '{{ customer.api_key }}', instanceUrl: '{{ base_url }}' }); // Verify a user is human lemma.verify().then(result => { if (result.verified) { // User is verified human - grant access showProtectedContent(); } else { // User couldn't be verified showVerificationNeeded(); } }); </script>

⚛️ React Integration

Using our React hooks for seamless integration:

1. Install the Package

npm install @lemma-network/verifier

2. Use the Hook

import { useLemmaVerification, LemmaGate } from '@lemma-network/verifier'; function MyComponent() { const { isVerified, verify, loading } = useLemmaVerification({ apiKey: '{{ customer.api_key }}', instanceUrl: '{{ base_url }}' }); if (loading) return <div>Checking verification...</div>; if (!isVerified) { return ( <button onClick={verify}> 🤖 Verify You're Human </button> ); } return ( <div> <h2>Welcome, verified human! 🎉</h2> {/* Your protected content here */} </div> ); } function App() { return ( <LemmaGate apiKey="{{ customer.api_key }}" instanceUrl="{{ base_url }}" fallback={<div>Please verify you're human</div>} > {/* This content only shows to verified humans */} <SecretContent /> </LemmaGate> ); }

3. Using the Gate Component

import { LemmaGate } from '@lemma-network/verifier'; function App() { return ( <LemmaGate apiKey="{{ customer.api_key }}" instanceUrl="{{ base_url }}" fallback={<div>Please verify you're human</div>} > {/* This content only shows to verified humans */} <SecretContent /> </LemmaGate> ); }

🚀 Express.js Backend

1. Install & Setup

3. Custom Verification Logic

const { verifyLemmaCredential } = require('@lemma-network/verifier'); app.post('/api/verify-user', async (req, res) => { const { credential } = req.body; const result = await verifyLemmaCredential(credential, { apiKey: '{{ customer.api_key }}', instanceUrl: '{{ base_url }}' }); if (result.verified) { // Grant access or create user session req.session.isHuman = true; res.json({ success: true, message: 'Human verified!' }); } else { res.status(403).json({ success: false, message: 'Verification failed' }); } });
