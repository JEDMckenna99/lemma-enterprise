JavaScript - Complete Auth Flow 
// Lemma Wallet-First Authentication for {{ customer.domain }}
// Passkey unlocks wallet → Credentials stored locally → No passwords!

const SITE_ID = '{{ customer.domain }}';

// Initialize wallet
const wallet = new LemmaWallet({ debug: true });

async function initAuth() {
    await wallet.init();
    
    // Check if user is already authenticated
    if (wallet.isAuthenticated()) {
        const credential = await wallet.getCredential('permission', SITE_ID);
        if (credential) {
            // User is signed in!
            const claims = credential.claims || credential.credentialSubject;
            showApp(claims);
            return;
        }
    }
    
    // Show sign-in button
    showSignInButton();
}

async function handleSignIn() {
    try {
        // Check if returning user (has passkey)
        const info = await wallet.getWalletInfo();
        
        if (info.hasPasskey) {
            // Returning user - unlock wallet
            await wallet.unlock();
        } else {
            // New user - register passkey
            await wallet.registerPasskey();
        }
        
        // Request permission for this site
        const walletSecret = await wallet.getWalletSecret();
        
        const response = await fetch('https://lemma.id/api/wallet-auth/issue', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                site_id: SITE_ID,
                wallet_secret: walletSecret
            })
        });
        
        const result = await response.json();
        
        if (result.success) {
            // Store credential in wallet
            await wallet.storeCredential(result.permission_lemma);
            
            // Reload to show authenticated state
            window.location.reload();
        }
    } catch (error) {
        console.error('Sign-in failed:', error);
        alert('Sign-in failed. Please try again.');
    }
}

function showSignInButton() {
    document.getElementById('auth-container').innerHTML = `
        <button onclick="handleSignIn()" class="signin-btn">
            Sign in with Passkey
        </button>
    `;
}

function showApp(userClaims) {
    document.getElementById('auth-container').innerHTML = `
        <div class="welcome">
            <p>Welcome back!</p>
            <button onclick="handleSignOut()">Sign Out</button>
        </div>
    `;
    // Your app code here...
}

async function handleSignOut() {
    const credentials = await wallet.getCredentials('permission');
    for (const cred of credentials) {
        const claims = cred.claims || cred.credentialSubject || {};
        if (claims.siteId === SITE_ID) {
            await wallet.removeCredential(cred.id);
        }
    }
    window.location.reload();
}

// Start authentication check
initAuth();

// React hook for Lemma auth - {{ customer.domain }}
import { useState, useEffect } from 'react';

const SITE_ID = '{{ customer.domain }}';

export function useLemmaAuth() {
    const [wallet, setWallet] = useState(null);
    const [isAuthenticated, setIsAuthenticated] = useState(false);
    const [user, setUser] = useState(null);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
        async function init() {
            const w = new window.LemmaWallet({ debug: true });
            await w.init();
            setWallet(w);
            
            if (w.isAuthenticated()) {
                const cred = await w.getCredential('permission', SITE_ID);
                if (cred) {
                    setIsAuthenticated(true);
                    setUser(cred.claims || cred.credentialSubject);
                }
            }
            setLoading(false);
        }
        init();
    }, []);

    const signIn = async () => {
        if (!wallet) return;
        const info = await wallet.getWalletInfo();
        info.hasPasskey ? await wallet.unlock() : await wallet.registerPasskey();
        
        const secret = await wallet.getWalletSecret();
        const res = await fetch('https://lemma.id/api/wallet-auth/issue', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({ site_id: SITE_ID, wallet_secret: secret })
        });
        const data = await res.json();
        if (data.success) {
            await wallet.storeCredential(data.permission_lemma);
            window.location.reload();
        }
    };

    const signOut = async () => {
        if (!wallet) return;
        const creds = await wallet.getCredentials('permission');
        for (const c of creds) {
            if ((c.claims?.siteId || c.credentialSubject?.siteId) === SITE_ID) {
                await wallet.removeCredential(c.id);
            }
        }
        window.location.reload();
    };

    return { isAuthenticated, user, loading, signIn, signOut };
}

// Usage:
function App() {
    const { isAuthenticated, user, loading, signIn, signOut } = useLemmaAuth();
    
    if (loading) return <div>Loading...</div>;
    
    if (!isAuthenticated) {
        return <button onClick={signIn}>Sign in with Passkey</button>;
    }
    
    return (
        <div>
            <p>Welcome!</p>
            <button onClick={signOut}>Sign Out</button>
        </div>
    );
}
