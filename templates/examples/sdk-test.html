Complete Integration Example

Copy this complete example to your site:

<!DOCTYPE html> <html> <head> <title>My Site with Lemma Auth</title> </head> <body> <div id="auth-container"></div> <div id="protected-content" style="display: none;"> <h1>Welcome!</h1> <p>You are signed in.</p> </div> <script src="https://lemma.id/static/js/lemma-signin-sdk.js"></script> <script> const auth = new LemmaSignIn({ siteId: 'YOUR_SITE_ID', onSignIn: (user) => { document.getElementById('protected-content').style.display = 'block'; document.getElementById('auth-container').innerHTML = '<p>Welcome, ' + user.email + '!</p>'; } }); auth.init().then(() => { if (!auth.isSignedIn()) { auth.showSignInButton('#auth-container'); } }); </script> </body> </html>
