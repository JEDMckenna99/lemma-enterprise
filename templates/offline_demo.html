🔧 Custom Integration

<script> const verifier = new LemmaOfflineVerifier({ enableLogging: true, strictMode: true }); async function checkUser() { const credential = JSON.parse(localStorage.getItem('lemma_credential') || '{}'); const result = await verifier.verify(credential); if (result.verified) { document.getElementById('protected-content').style.display = 'block'; console.log(`Verified in ${result.verification_time_ms}ms with ${result.network_calls} network calls`); } else { // Redirect to verification window.location.href = 'https://lemma-enterprise-0f6ba17076c1.herokuapp.com/verify'; } } checkUser(); </script>

⚡ Advanced Integration with Callbacks

<script> const simpleVerifier = new LemmaSimpleVerifier({ enableLogging: true, onVerified: (result) => { console.log('User verified:', result); showProtectedContent(); updateUI({ verified: true, time: result.verification_time_ms, method: result.verification_method }); }, onFailed: (error) => { console.log('Verification failed:', error); showVerificationPrompt(); } }); // Auto-verify when page loads simpleVerifier.autoVerify(); </script>
