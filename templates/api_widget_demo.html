HTML Integration

<!-- Include Lemma Widget Scripts -->
<script src="https://lemma.id/static/js/lemma-shield-widget.js"></script>
<link rel="stylesheet" href="https://lemma.id/static/css/lemma-shield-widget.css">

<!-- Protected Content -->
<div id="protected-content">
  <h3>Premium Content</h3>
  <p>This content requires verification</p>
</div>

JavaScript Configuration

// Initialize Modern Lemma Widget
const lemmaWidget = new LemmaShieldWidget({
  protectedContent: '#protected-content',
  securityLevel: 'enterprise',
  theme: 'modern-saas',
  onVerified: () => {
    console.log('✅ User verified with modern widget!');
    showProtectedContent();
  },
  onError: (error) => {
    console.error('❌ Verification error:', error);
  }
});
