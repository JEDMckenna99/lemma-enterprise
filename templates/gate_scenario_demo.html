Implementation Guide

1. Basic Gate Setup

<!-- Include Lemma Gate -->
<script src="https://your-lemma-instance.com/static/js/lemma-gate.js"></script>

<!-- Protected Content -->
<div data-lemma-gate="true" data-scenario="premium-content">
  <h2>Premium Article</h2>
  <p>This content requires human verification to access.</p>
</div>

2. Advanced Configuration

// Configure gate behavior
window.lemmaGateConfig = {
  scenario: 'premium-content',
  fallbackMessage: 'Please verify you are human to continue',
  redirectUrl: '/verify',
  allowOffline: true,
  cacheCredentials: true
};

3. Backend Verification

// Express.js middleware example
app.use('/premium/*', async (req, res, next) => {
  const credential = req.headers['x-lemma-credential'];
  
  if (await verifyLemmaCredential(credential)) {
    next(); // User is verified human
  } else {
    res.status(401).json({ error: 'Human verification required' });
  }
});
