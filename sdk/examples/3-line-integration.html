Advanced Usage (Optional):

🔐 Manual Verification (Identity Creation)

// Trigger verification flow for new users
Lemma.verify();  // Redirects to Stripe KYC

// Listen for verification completion
window.addEventListener('lemma:verified', function(event) {
    console.log('New identity credential:', event.detail.credential);
});

⚡ Manual Authentication (Access Control)

// Check authentication status (microsecond response)
const auth = await Lemma.authenticate();
console.log('Authenticated:', auth.verified, 'Time:', auth.timing + 'µs');

// Manual protection control
Lemma.protect('#specific-element');  // Protect specific elements
console.log(Lemma.status());         // Check federation status

🔄 Event Handling

// Listen for ready event
window.addEventListener('lemma:ready', function(event) {
    console.log('Lemma federation ready!', event.detail);
});

// Listen for authentication updates
window.addEventListener('lemma:auth-update', function(event) {
    console.log('Auth status changed:', event.detail.verified);
});
