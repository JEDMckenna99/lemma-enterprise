Basic Setup

// Import the SDK
import { Lemma } from '@lemma/verification-sdk';

// Initialize
const lemma = new Lemma({
    apiKey: 'your-api-key',
    debug: true
});

// Wait for ready
lemma.on('ready', () => {
    console.log('Lemma SDK ready!');
});

// Verify a credential
const result = await lemma.verify(credentialData);
console.log('Verification result:', result);

React Integration

// React component
import React, { useState, useEffect } from 'react';
import { Lemma } from '@lemma/verification-sdk';

function CredentialVerifier() {
    const [lemma] = useState(() => new Lemma());
    const [result, setResult] = useState(null);
    const [loading, setLoading] = useState(false);
    
    const handleVerify = async (credentialData) => {
        setLoading(true);
        try {
            const result = await lemma.verify(credentialData);
            setResult(result);
        } catch (error) {
            console.error('Verification failed:', error);
        } finally {
            setLoading(false);
        }
    };
    
    return (
        <div>
            <button onClick={() => handleVerify(credentialData)}>
                {loading ? 'Verifying...' : 'Verify Credential'}
            </button>
            {result && (
                <div className={result.verified ? 'success' : 'error'}>
                    {result.verified ? '✅ Verified' : '❌ Invalid'}
                </div>
            )}
        </div>
    );
}
